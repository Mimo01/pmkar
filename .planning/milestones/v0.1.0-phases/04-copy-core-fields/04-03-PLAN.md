```rust
#[tauri::command]
pub async fn fetch_cloud_meta(
    base_url: String,
    db: State<'_, Arc<Mutex<AuditDb>>>,
) -> Result<CloudMeta, AppError> {
    let client = build_audited_client(db.inner().clone())?;

    // Get credentials from keychain
    let email = crate::commands::get_credential(
        "jira_cloud".to_string(),
        "email".to_string(),
    )?;
    let api_token = crate::commands::get_credential(
        "jira_cloud".to_string(),
        "api_token".to_string(),
    )?;

    let auth = format!(
        "Basic {}",
        base64::engine::general_purpose::STANDARD.encode(format!("{}:{}", email, api_token))
    );

    // Fetch current user
    let myself_resp = client
        .get(format!("{}/rest/api/3/myself", base_url))
        .header("Authorization", &auth)
        .send()
        .await
        .map_err(|e| AppError::Network(format!("Failed to fetch /myself: {}", e)))?;
    let myself_body: serde_json::Value = myself_resp
        .json()
        .await
        .map_err(|e| AppError::Network(format!("Failed to parse /myself: {}", e)))?;
    let account_id = myself_body["accountId"]
        .as_str()
        .unwrap_or("")
        .to_string();

    // Fetch priorities
    let prio_resp = client
        .get(format!("{}/rest/api/3/priority", base_url))
        .header("Authorization", &auth)
        .send()
        .await
        .map_err(|e| AppError::Network(format!("Failed to fetch /priority: {}", e)))?;
    let prio_body: serde_json::Value = prio_resp
        .json()
        .await
        .map_err(|e| AppError::Network(format!("Failed to parse /priority: {}", e)))?;
    let priorities: Vec<PriorityOption> = prio_body
        .as_array()
        .unwrap_or(&vec![])
        .iter()
        .filter_map(|p| {
            Some(PriorityOption {
                id: p["id"].as_str()?.to_string(),
                name: p["name"].as_str()?.to_string(),
            })
        })
        .collect();

    // Fetch project statuses — use hardcoded project key for now
    let status_resp = client
        .get(format!("{}/rest/api/3/project/MYPROJ/statuses", base_url))
        .header("Authorization", &auth)
        .send()
        .await
        .map_err(|e| AppError::Network(format!("Failed to fetch project statuses: {}", e)))?;
    let status_body: serde_json::Value = status_resp
        .json()
        .await
        .map_err(|e| AppError::Network(format!("Failed to parse project statuses: {}", e)))?;
    let statuses: Vec<StatusOption> = status_body
        .as_array()
        .and_then(|arr| arr.first())
        .and_then(|first| first["statuses"].as_array())
        .unwrap_or(&vec![])
        .iter()
        .filter_map(|s| {
            Some(StatusOption {
                id: s["id"].as_str()?.to_string(),
                name: s["name"].as_str()?.to_string(),
            })
        })
        .collect();

    Ok(CloudMeta {
        available_statuses: statuses,
        available_priorities: priorities,
        current_account_id: account_id,
        cloud_base_url: base_url,
    })
}
```

3. Add the `copy_ticket` command. The pipeline implements D-03's two-pass approach:

   a. **Step 1: Fetch source issue with renderedFields** — `GET {source_base_url}/rest/api/2/issue/{source_key}?expand=renderedFields` using Server credentials (from keychain, same pattern as fetch_ticket_detail).

   b. **Step 2: Extract rendered HTML description** — `source_body["renderedFields"]["description"]` as string. Also extract source summary from `source_body["fields"]["summary"]`.

   c. **Step 3: Collect inline image URLs** — Find all `<img` tags in the HTML using regex (`r#"<img[^>]+src="([^"]+)""#`). For each image URL, extract filename from the URL path. Store as `Vec<(String, String)>` of (old_url, filename) pairs.

   d. **Step 4: Create issue in Cloud with placeholder description** — First, create the issue WITHOUT the description (or with a minimal placeholder ADF). This is needed because we cannot upload attachments until the issue exists, and we need the attachment URLs to rewrite the HTML before ADF conversion.
      `POST {target_base_url}/rest/api/3/issue` with body:
      ```json
      {
        "fields": {
          "project": { "key": "MYPROJ" },
          "issuetype": { "name": "Task" },
          "summary": target_summary,
          "priority": { "id": target_priority_id },
          "labels": target_labels,
          "assignee": { "accountId": current_account_id }
        }
      }
      ```
      Record step result. If fails, return early with failure result.
      Extract `target_key` from response (`key` field).

   e. **Step 5: Upload inline images and build URL map** (per D-03) — For each image found in step c:
      - Download image bytes from source Jira using Server PAT auth
      - Upload to target ticket: `POST {target_base_url}/rest/api/3/issue/{target_key}/attachments` with `X-Atlassian-Token: no-check` header and multipart form
      - Extract the `content` URL from the upload response (the response is a JSON array; the first element has a `content` field with the new attachment URL)
      - Add to url_map: `old_url -> new_content_url`
      - Record per-image step result
      If an individual image upload fails, record the failure but continue with remaining images.

   f. **Step 6: Rewrite HTML image URLs and convert to ADF** (per D-03) — Call `rewrite_image_urls(&html, &url_map)` to replace all source image URLs with the new Cloud attachment URLs in the HTML. Then call `htmltoadf::convert_html_str_to_adf_str(&rewritten_html)` and parse result with `serde_json::from_str()`.

   g. **Step 7: Update issue description with rewritten ADF** — `PUT {target_base_url}/rest/api/3/issue/{target_key}` with body:
      ```json
      {
        "fields": {
          "description": adf_value
        }
      }
      ```
      Record step result.

   h. **Step 8: Add remote link** — `POST {target_base_url}/rest/api/3/issue/{target_key}/remotelink` with body:
      ```json
      {
        "globalId": "pmkar-source={urlencoded_source_base_url}&key={source_key}",
        "object": {
          "url": "{source_base_url}/browse/{source_key}",
          "title": "{source_key}: {source_summary}"
        },
        "relationship": "copied from"
      }
      ```
      Record step result.
