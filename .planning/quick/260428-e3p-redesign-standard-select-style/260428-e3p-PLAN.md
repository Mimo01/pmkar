1. Add import at the top of the file alongside existing feature-renderer imports:
   ```typescript
   import { VirtualizedCombobox } from '@/features/field-renderers/components/VirtualizedCombobox';
   ```

2. Locate the `<select id="copy-target-project-modal" ...>` block (around line 334). Replace the entire `<select>...</select>` element (including all `<option>` children) with:
   ```tsx
   <VirtualizedCombobox<{ key: string; name: string }>
     items={cloudProjects}
     value={cloudProjects.find(p => p.key === targetProjectKey) ?? null}
     onChange={(p) => setTargetProjectKey(p.key)}
     displayLabel={(p) => `${p.name} (${p.key})`}
     filterFn={(p, q) =>
       p.name.toLowerCase().includes(q.toLowerCase()) ||
       p.key.toLowerCase().includes(q.toLowerCase())
     }
     placeholder={t('settings.project.select')}
     ariaLabel={t('copy.targetProject')}
   />
   ```

3. Remove the `id="copy-target-project-modal"` attribute — VirtualizedCombobox manages its own internal ids. The wrapping `<label>` (if any) must use `aria-label` on the combobox itself, which is already covered by the `ariaLabel` prop.

4. Do NOT remove the label element above the select — just the select itself.
