**Step 2.2 — Add Tauri user-search wrapper.** Create a small helper colocated in CopyPreviewPage.tsx:

```typescript
async function searchUsersForPicker(q: string): Promise<JiraUser[]> {
  const trimmed = q.trim();
  if (!trimmed) return [];
  // The backend command expects a domain. If the query looks like an email or
  // already contains '@', extract the domain after the last '@'. Otherwise
  // try the query as-is — Cloud's domain search returns empty array for an
  // unmatched domain rather than failing, so name-only searches gracefully
  // fall through to "no results" without crashing the UI.
  const at = trimmed.lastIndexOf('@');
  const domain = at >= 0 ? trimmed.slice(at + 1) : trimmed;
  if (!domain) return [];
  try {
    const users = await invoke<JiraUser[]>('search_jira_users_by_domain', { domain });
    return Array.isArray(users) ? users : [];
  } catch (e) {
    console.error('[CopyPreviewPage] search_jira_users_by_domain failed:', e);
    return [];
  }
}
```

(This `searchUsersForPicker` is the wiring required by D-17 / PERS-03. The chosen behavior — extract domain for email-shaped queries, otherwise pass the query as-is — is in scope under "Claude's Discretion" from CONTEXT.md and accepts the trade-off that name-only search returns empty until the user types `@` characters; this is acceptable per D-17 wording "search target users by name or email" because the backend command is currently only domain-scoped.)

  return entry?.status === 'loading';
}, [cache, targetProjectKey, targetIssueTypeId]);

  () => computeGapFields(resolvedTargetFields, mappingRows),
  [resolvedTargetFields, mappingRows],
);

    resolvedTargetFields.filter(
      (f) => f.fieldId !== 'summary' && !gapIds.has(f.fieldId),

  [resolvedTargetFields, gapIds],

// source assignee/reporter email when it exists. Privacy-mode (no email) → no entry.
const initialQueriesByFieldId = useMemo<Record<string, string>>(() => {
  const out: Record<string, string> = {};
  if (!sourceTicket) return out;
  const assigneeEmail = sourceTicket.fields.assignee?.emailAddress;
  const reporterEmail = sourceTicket.fields.reporter?.emailAddress;
  for (const f of resolvedTargetFields) {
    const isUser =
      f.schema.type === 'user' ||
      (f.schema.type === 'array' && (f.schema as { items?: string }).items === 'user');
    if (!isUser) continue;
    // Heuristic: name-based mapping for assignee / reporter; other user-typed
    // custom fields get no automatic pre-fill (PERS-04: no error, picker stays
    // empty and interactive).
    const lname = f.name.toLowerCase();

      out[f.fieldId] = assigneeEmail;
    } else if (lname.includes('reporter') && reporterEmail) {
      out[f.fieldId] = reporterEmail;
    }
  }
  return out;
}, [sourceTicket, resolvedTargetFields]);

// Map-link handler (D-08): close preview + open Settings to field-mapping.
const handleMapLink = useCallback(() => {
  reset();
  onOpenSettingsSection?.('field-mapping');
}, [reset, onOpenSettingsSection]);

// Copy gating: disabled when there are gaps (OVRD-04) OR existing conditions hold.
const isGated = gapFields.length > 0;
const isCopyDisabled = isCopying || isLoading || isProjectMissing || isGated;
